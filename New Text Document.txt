local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")

-- ===== CONFIG =====
local ENABLE_SEND_TRADE = false       -- set false to disable sending trade requests
local ENABLE_ACCEPT_DECLINE = true   -- set false to disable auto-accepting trade requests

local TARGET_PLAYERS = {
	"Her0eXBuildera31",
}

local RETRY_INTERVAL = 5 -- seconds between retry attempts if remote isn't found yet
-- ===================

local tradeRemote = nil
local acceptDeclineRemote = nil

local function sendTradeRequest(playerName)
	local targetPlayer = Players:FindFirstChild(playerName)
	if not targetPlayer then
		warn("Player not found in server: " .. playerName)
		return
	end

	if tradeRemote and tradeRemote:IsA("RemoteEvent") then
		tradeRemote:FireServer(targetPlayer)
		print("Successfully sent trade request to: " .. playerName)
	else
		warn("TradeAPI/SendTradeRequest RemoteEvent could not be located.")
	end
end

local function onTradeRequest(fromPlayer, ...)
	if acceptDeclineRemote and acceptDeclineRemote:IsA("RemoteEvent") then
		acceptDeclineRemote:FireServer(fromPlayer, true, ...) -- true = accept
		print("Auto-accepted trade request from: " .. (fromPlayer and fromPlayer.Name or "unknown"))
	end
end

-- Loop that keeps trying to locate + hook up the Send Trade remote
if ENABLE_SEND_TRADE then
	task.spawn(function()
		while true do
			if not tradeRemote then
				tradeRemote = ReplicatedStorage:FindFirstChild("TradeAPI/SendTradeRequest", true)
			end

			if tradeRemote and tradeRemote:IsA("RemoteEvent") then
				for _, playerName in ipairs(TARGET_PLAYERS) do
					if Players:FindFirstChild(playerName) then
						sendTradeRequest(playerName)
					end
				end
			else
				warn("TradeAPI/SendTradeRequest not found, retrying in " .. RETRY_INTERVAL .. "s...")
			end

			task.wait(RETRY_INTERVAL)
		end
	end)
end

-- Loop that keeps trying to locate + hook up the Accept/Decline remote
if ENABLE_ACCEPT_DECLINE then
	task.spawn(function()
		while true do
			if not acceptDeclineRemote then
				acceptDeclineRemote = ReplicatedStorage:FindFirstChild("TradeAPI/AcceptOrDeclineTradeRequest", true)

				if acceptDeclineRemote and acceptDeclineRemote:IsA("RemoteEvent") then
					acceptDeclineRemote.OnClientEvent:Connect(onTradeRequest)
					print("Connected to TradeAPI/AcceptOrDeclineTradeRequest")
				else
					warn("TradeAPI/AcceptOrDeclineTradeRequest not found, retrying in " .. RETRY_INTERVAL .. "s...")
				end
			end

			task.wait(RETRY_INTERVAL)
		end
	end)
end
