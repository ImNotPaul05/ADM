local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")

-- Configuration: Put the exact Roblox player names you want to target
local TARGET_PLAYERS = {
	"",
}

-- Locate the RemoteEvents
local sendTradeRemote = ReplicatedStorage:FindFirstChild("TradeAPI/SendTradeRequest", true)
local acceptDeclineRemote = ReplicatedStorage:FindFirstChild("TradeAPI/AcceptOrDeclineTradeRequest", true)

local function processTradeForPlayer(playerName)
	local targetPlayer = Players:FindFirstChild(playerName)
	if not targetPlayer then
		warn("Player not found in server: " .. playerName)
		return
	end
	
	-- Example 1: Send a trade request
	if sendTradeRemote and sendTradeRemote:IsA("RemoteEvent") then
		sendTradeRemote:FireServer(targetPlayer)
		print("Sent trade request to: " .. playerName)
	end
	
	-- Example 2: Accept or decline a request (passing true to accept, or false to decline depending on game logic)
	if acceptDeclineRemote and acceptDeclineRemote:IsA("RemoteEvent") then
		-- Depending on the game, you may need to pass the target player or a specific request ID/boolean
		acceptDeclineRemote:FireServer(targetPlayer, true)
		print("Processed Accept/Decline action for: " .. playerName)
	end
end

-- Iterate through the configuration list
for _, playerName in ipairs(TARGET_PLAYERS) do
	if Players:FindFirstChild(playerName) then
		processTradeForPlayer(playerName)
	end
end
