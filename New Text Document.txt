local Players = game:GetService("Players")

local function triggerTradeForPlayer(playerName)
	local player = Players:FindFirstChild(playerName)
	if not player then return end
	
	-- Use WaitForChild in case PlayerGui or the App is still loading
	local playerGui = player:WaitForChild("PlayerGui", 5)
	if not playerGui then return end
	
	local profileApp = playerGui:WaitForChild("PlayerProfileApp", 5)
	if not profileApp then 
		print("PlayerProfileApp not found for " .. playerName)
		return 
	end
	
	-- Safely trace down the path using WaitForChild
	local frame = profileApp:WaitForChild("Frame", 5)
	local header = frame and frame:WaitForChild("Header", 5)
	local tradeButton = header and header:WaitForChild("TradeButton", 5)
	
	if tradeButton then
		print("Successfully located TradeButton for " .. playerName)
		-- Trigger your logic here
	else
		warn("TradeButton does not exist anywhere in PlayerGui for " .. playerName)
	end
end

-- Example usage:
triggerTradeForPlayer("Her0eXBuildera31")
