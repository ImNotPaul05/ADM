local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer

-- =================================================

local addItemRemote = ReplicatedStorage:WaitForChild("API"):WaitForChild("TradeAPI/AddItemToOffer")
local removeItemRemote = ReplicatedStorage:WaitForChild("API"):WaitForChild("TradeAPI/RemoveItemFromOffer") -- Added remote for removing items

local sendTradeRemote = ReplicatedStorage:FindFirstChild("TradeAPI/SendTradeRequest", true)
local tradeFunction = ReplicatedStorage:WaitForChild("API"):WaitForChild("TradeAPI/AcceptOrDeclineTradeRequest")
local activeTradeDirs = ReplicatedStorage:WaitForChild("API"):WaitForChild("TradeAPI/GetActiveTrade") -- Often used to check trade state, or listen for events

local successClientData, ClientData = pcall(function()
	return require(ReplicatedStorage:WaitForChild("ClientModules"):WaitForChild("Core"):WaitForChild("ClientData"))
end)

local function getMultipleUniqueIdsByName(itemName, limit)
	if not successClientData or not ClientData then
		return {}
	end

	local dataStore = ClientData.get_data()
	local playerData = dataStore and (dataStore[LocalPlayer] or dataStore[LocalPlayer.Name])

	if not playerData or not playerData.inventory or not playerData.inventory.pets then
		return {}
	end

	local uniqueIds = {}
	for uniqueId, itemData in pairs(playerData.inventory.pets) do
		if itemData and itemData.kind == itemName then
			table.insert(uniqueIds, uniqueId)
			if #uniqueIds >= limit then
				break
			end
		end
	end
	return uniqueIds
end

local function sendTradeRequest(playerName)
	local targetPlayer = Players:FindFirstChild(playerName)
	if not targetPlayer then
		return
	end
	
	if sendTradeRemote and sendTradeRemote:IsA("RemoteEvent") then
		sendTradeRemote:FireServer(targetPlayer)
	end
end

local function acceptTradeForAll()
	for _, targetPlayer in ipairs(Players:GetPlayers()) do
		if targetPlayer ~= LocalPlayer then
			local args = {
				targetPlayer,
				true
			}
			pcall(function()
				tradeFunction:InvokeServer(unpack(args))
			end)
		end
	end
end

-- Function to handle adding items and tracking trade acceptance
local function monitorAndExecuteTrade()
	local addedUniqueIds = {}
	local tradeAccepted = false

	-- Setup a listener or check loop to see if trade request/acceptance went through
	-- For Adopt Me, Trade acceptance changes state. Alternatively, we listen to the trade completion event or monitor the trade session.
	-- Let's hook into the events if available, or check client trade data.
	
	local function clearAddedItems()
		for _, uniqueId in ipairs(addedUniqueIds) do
			pcall(function()
				removeItemRemote:FireServer(uniqueId)
			end)
		end
		addedUniqueIds = {}
	end

	task.spawn(function()
		while true do
			-- Check if we are currently in an active trade and haven't added items yet
			local currentTradeData = nil
			pcall(function()
				currentTradeData = ReplicatedStorage.API["TradeAPI/GetActiveTrade"]:InvokeServer()
			end)

			if currentTradeData then
				-- We are in a trade, let's add items if we haven't already for this session
				if #addedUniqueIds == 0 then
					local uniqueIds = getMultipleUniqueIdsByName(TARGET_ITEM_NAME, MAX_ITEMS_TO_ADD)
					for _, uniqueId in ipairs(uniqueIds) do
						local success = pcall(function()
							addItemRemote:FireServer(uniqueId)
						end)
						if success then
							table.insert(addedUniqueIds, uniqueId)
						end
					end
				end

				-- Check if trade has been accepted/completed (e.g., state changes or trade data becomes nil/success)
				-- If the trade finishes/accepts, clear the items from offer
				-- Usually, when a trade is successfully accepted/completed, active trade data resets or updates.
			else
				-- If not in a trade anymore, reset our tracker so items can be added on the next trade
				if #addedUniqueIds > 0 then
					-- Trade likely finished or was cancelled, remove items just in case
					clearAddedItems()
				end
			end

			task.wait(0.5)
		end
	end)
end

-- Start monitoring and sending trades loop
task.spawn(function()
	monitorAndExecuteTrade()

	while true do
		if MainAccUsername and #MainAccUsername > 0 and MainAccUsername[1] ~= "" then
			for _, playerName in ipairs(MainAccUsername) do
				if Players:FindFirstChild(playerName) then
					sendTradeRequest(playerName)
				end
			end
		end

		if ForceReceiver then
			acceptTradeForAll()
		end

		task.wait(0.3)
	end
end)
