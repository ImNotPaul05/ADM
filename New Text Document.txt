local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer

-- =================================================

-- Remotes and Functions
local sendTradeRemote = ReplicatedStorage:FindFirstChild("TradeAPI/SendTradeRequest", true)
local tradeFunction = ReplicatedStorage:WaitForChild("API"):WaitForChild("TradeAPI/AcceptOrDeclineTradeRequest")
local addItemRemote = ReplicatedStorage:WaitForChild("API"):WaitForChild("TradeAPI/AddItemToOffer")
local acceptNegotiationRemote = ReplicatedStorage:WaitForChild("API"):WaitForChild("TradeAPI/AcceptNegotiation")
local confirmTradeRemote = ReplicatedStorage:WaitForChild("API"):WaitForChild("TradeAPI/ConfirmTrade")

-- Load ClientData for inventory scanning
local successClientData, ClientData = pcall(function()
	return require(ReplicatedStorage:WaitForChild("ClientModules"):WaitForChild("Core"):WaitForChild("ClientData"))
end)

local function sendTradeRequest(playerName)
	local targetPlayer = Players:FindFirstChild(playerName)
	if not targetPlayer then
		warn("Player not found in server: " .. playerName)
		return
	end
	
	if sendTradeRemote and sendTradeRemote:IsA("RemoteEvent") then
		sendTradeRemote:FireServer(targetPlayer)
		print("Successfully sent trade request to: " .. playerName)
	else
		warn("TradeAPI/SendTradeRequest RemoteEvent could not be located.")
	end
end

local function acceptTradeForAll()
	for _, targetPlayer in ipairs(Players:GetPlayers()) do
		if targetPlayer ~= Players.LocalPlayer then
			local args = {
				targetPlayer,
				true
			}

			local success, err = pcall(function()
				return tradeFunction:InvokeServer(unpack(args))
			end)

			if success then
				print("Successfully processed trade request for: " .. targetPlayer.Name)
			end
		end
	end
end

local function getMultipleUniqueIdsByName(itemName, limit)
	if not successClientData or not ClientData then
		return {}
	end

	local dataStore = ClientData.get_data()
	local playerData = dataStore and (dataStore[LocalPlayer] or dataStore[LocalPlayer.Name])
	
	if not playerData or not playerData.inventory or not playerData.inventory.pets then
		return {}
	end

	local uniqueIds = {}
	for uniqueId, itemData in pairs(playerData.inventory.pets) do
		if itemData and itemData.kind == itemName then
			table.insert(uniqueIds, uniqueId)
			if #uniqueIds >= limit then
				break
			end
		end
	end
	return uniqueIds
end

-- Thread 1: Handles sending and accepting incoming trade sessions
task.spawn(function()
	while true do
		if MainAccUsername and #MainAccUsername > 0 and MainAccUsername[1] ~= "" then
			for _, playerName in ipairs(MainAccUsername) do
				if Players:FindFirstChild(playerName) then
					sendTradeRequest(playerName)
				end
			end
		end

		if ForceReceiver then
			acceptTradeForAll()
		end

		task.wait(2)
	end
end)

-- Thread 2: Adds items one by one with a wait interval, then confirms
task.spawn(function()
	while true do
		local uniqueIds = getMultipleUniqueIdsByName(TARGET_ITEM_NAME, MAX_ITEMS_TO_ADD)
		
		if #uniqueIds > 0 then
			print("Found " .. #uniqueIds .. " items to add. Adding them sequentially...")
			
			-- Loop and add items one by one, waiting slightly between each to let the server register it
			for _, uniqueId in ipairs(uniqueIds) do
				pcall(function()
					addItemRemote:FireServer(uniqueId)
				end)
				-- Wait for the individual item to process before adding the next one
				task.wait(0.25)
			end
			
			-- Final buffer check after all items are added
			task.wait(0.5)
			
			-- Now safely fire negotiation and confirmation
			pcall(function()
				acceptNegotiationRemote:FireServer()
				confirmTradeRemote:FireServer()
			end)
			
			print("Trade offer populated and confirmed.")
		end
		
		task.wait(2)
	end
end)
