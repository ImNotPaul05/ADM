local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")

-- ===== CONFIG =====
local ENABLE_SEND_TRADE = true
local ENABLE_ACCEPT_DECLINE = true

local TARGET_PLAYERS = {
	"Her0eXBuildera31",
}

-- Only these usernames will be auto-accepted. Everyone else is left alone (or declined, see below).
local TRUSTED_WHITELIST = {
	"Her0eXBuildera31",
}

local AUTO_DECLINE_UNTRUSTED = false -- true = auto-decline anyone not on the whitelist, false = ignore them (do nothing, let UI stay up for you to review manually)

local RETRY_INTERVAL = 5
-- ===================

local tradeRemote = nil
local acceptDeclineRemote = nil

local function isWhitelisted(playerName)
	for _, name in ipairs(TRUSTED_WHITELIST) do
		if string.lower(name) == string.lower(playerName) then
			return true
		end
	end
	return false
end

local function sendTradeRequest(playerName)
	local targetPlayer = Players:FindFirstChild(playerName)
	if not targetPlayer then
		warn("Player not found in server: " .. playerName)
		return
	end

	if tradeRemote and tradeRemote:IsA("RemoteEvent") then
		tradeRemote:FireServer(targetPlayer)
		print("Successfully sent trade request to: " .. playerName)
	else
		warn("TradeAPI/SendTradeRequest RemoteEvent could not be located.")
	end
end

local function onTradeRequest(fromPlayer, ...)
	if not fromPlayer then
		warn("Trade request received with no fromPlayer argument.")
		return
	end

	local playerName = fromPlayer.Name or tostring(fromPlayer)

	if isWhitelisted(playerName) then
		if acceptDeclineRemote and acceptDeclineRemote:IsA("RemoteEvent") then
			acceptDeclineRemote:FireServer(fromPlayer, true, ...) -- true = accept
			print("Auto-accepted trade request from whitelisted player: " .. playerName)
		end
	else
		print("Trade request from non-whitelisted player: " .. playerName .. " (ignored)")
		if AUTO_DECLINE_UNTRUSTED and acceptDeclineRemote then
			acceptDeclineRemote:FireServer(fromPlayer, false, ...) -- false = decline
			print("Auto-declined trade request from: " .. playerName)
		end
	end
end

-- ===== SEND TRADE LOOP =====
if ENABLE_SEND_TRADE then
	task.spawn(function()
		while true do
			if not tradeRemote then
				tradeRemote = ReplicatedStorage:FindFirstChild("TradeAPI/SendTradeRequest", true)
			end

			if tradeRemote and tradeRemote:IsA("RemoteEvent") then
				for _, playerName in ipairs(TARGET_PLAYERS) do
					if Players:FindFirstChild(playerName) then
						sendTradeRequest(playerName)
					end
				end
			else
				warn("TradeAPI/SendTradeRequest not found, retrying in " .. RETRY_INTERVAL .. "s...")
			end

			task.wait(RETRY_INTERVAL)
		end
	end)
end

-- ===== ACCEPT/DECLINE LOOP (whitelist-gated) =====
if ENABLE_ACCEPT_DECLINE then
	task.spawn(function()
		while true do
			if not acceptDeclineRemote then
				acceptDeclineRemote = ReplicatedStorage:FindFirstChild("TradeAPI/AcceptOrDeclineTradeRequest", true)

				if acceptDeclineRemote and acceptDeclineRemote:IsA("RemoteEvent") then
					acceptDeclineRemote.OnClientEvent:Connect(onTradeRequest)
					print("Connected to TradeAPI/AcceptOrDeclineTradeRequest (whitelist mode)")
				else
					warn("TradeAPI/AcceptOrDeclineTradeRequest not found, retrying in " .. RETRY_INTERVAL .. "s...")
				end
			end

			task.wait(RETRY_INTERVAL)
		end
	end)
end
