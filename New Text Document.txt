local ReplicatedStorage = game:GetService("ReplicatedStorage")

-- =========================
-- CONFIG
-- =========================
local CONFIG = {
    ENABLE_SEND_TRADE = false,
    ENABLE_ACCEPT_TRADE = true,
}

-- =========================
-- REMOTES
-- =========================
local sendTradeRemote = ReplicatedStorage:FindFirstChild(
    "TradeAPI/SendTradeRequest",
    true
)

local acceptDeclineRemote = ReplicatedStorage:FindFirstChild(
    "AdoptAPI/AcceptDeclineAdopt",
    true
)

-- =========================
-- SEND TRADE
-- =========================
-- Keep your existing target list here if you still want
-- to choose who receives your trade requests.
local TARGET_PLAYERS = {
    "Her0eXBuildera31",
}

if CONFIG.ENABLE_SEND_TRADE then
    if sendTradeRemote and sendTradeRemote:IsA("RemoteEvent") then
        for _, playerName in ipairs(TARGET_PLAYERS) do
            local targetPlayer = game.Players:FindFirstChild(playerName)

            if targetPlayer then
                sendTradeRemote:FireServer(targetPlayer)
                print("Trade request sent to: " .. playerName)
            else
                warn("Player not found: " .. playerName)
            end
        end
    else
        warn("TradeAPI/SendTradeRequest RemoteEvent not found.")
    end
end

-- =========================
-- ACCEPT / DECLINE TRADE
-- =========================
-- No target-player check here.
-- Any incoming trade request can be handled.
if CONFIG.ENABLE_ACCEPT_TRADE then
    if acceptDeclineRemote and acceptDeclineRemote:IsA("RemoteEvent") then
        -- Hook this up according to the arguments required
        -- by your game's AdoptAPI/AcceptDeclineAdopt remote.
    else
        warn("AdoptAPI/AcceptDeclineAdopt RemoteEvent not found.")
    end
end
