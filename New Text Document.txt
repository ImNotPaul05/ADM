local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")

-- Configuration
local CONFIG = {
	TargetPlayers = {
		"Her0eXBuildera31",
		-- Add more usernames here as needed
	},
	AutoSendTrade = false,     -- Set to true to send trade requests, false to disable
	AutoAcceptTrade = true,   -- Set to true to accept trade requests, false to disable
}

-- Locate the RemoteEvents
local sendTradeRemote = ReplicatedStorage:FindFirstChild("TradeAPI/SendTradeRequest", true)
local acceptTradeRemote = ReplicatedStorage:FindFirstChild("TradeAPI/AcceptOrDeclineTradeRequest", true)

local function processTradeForPlayer(playerName)
	local targetPlayer = Players:FindFirstChild(playerName)
	if not targetPlayer then
		warn("Player not found in server: " .. playerName)
		return
	end
	
	-- Send the trade request if enabled in config
	if CONFIG.AutoSendTrade and sendTradeRemote and sendTradeRemote:IsA("RemoteEvent") then
		sendTradeRemote:FireServer(targetPlayer)
		print("Sent trade request to: " .. playerName)
	end
	
	-- Accept trade interaction if enabled in config
	if CONFIG.AutoAcceptTrade and acceptTradeRemote and acceptTradeRemote:IsA("RemoteEvent") then
		acceptTradeRemote:FireServer(targetPlayer, true)
		print("Accepted trade interaction for: " .. playerName)
	end
end

-- Iterate through the configuration list
for _, playerName in ipairs(CONFIG.TargetPlayers) do
	if Players:FindFirstChild(playerName) then
		processTradeForPlayer(playerName)
	end
end

-- Automatically process if they join later
Players.PlayerAdded:Connect(function(player)
	for _, playerName in ipairs(CONFIG.TargetPlayers) do
		if player.Name == playerName then
			task.wait(2)
			processTradeForPlayer(playerName)
		end
	end
end)
